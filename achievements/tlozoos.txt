# The linked-game arc. wIsLinkedGame ($cc01) is live during play; the
# global flags ($c6ca + GLOBALFLAG/8, bit GLOBALFLAG%8) are on the file,
# so end-of-game flags unlock the next time that file is played.

[linked-quest]

# tracks the four hidden golden beasts, one bit each.

# unlocks happen.)
